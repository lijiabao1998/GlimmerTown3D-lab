// 批次拍天氣疊層：node run-wx.mjs [viewIds]
import { spawnSync } from 'node:child_process';
import fs from 'node:fs';
const SP = '/tmp/claude-0/-home-user-GlimmerTown3D-lab/1669b73f-8ee1-581b-9f53-351901c7277e/scratchpad';
const OUT = SP + '/d063/wx';
fs.mkdirSync(OUT, { recursive: true });
const VIEWS = [
  { id: 'seed516-far', sample: 'seed516', x: 36, z: 36, zoom: 1.2, W: 1440, H: 1000, mob: 0 },
  { id: 'seed516-district', sample: 'seed516', x: 40.4, z: 6.7, zoom: 2.56, W: 412, H: 860, mob: 1 },
  { id: 'newcity-far', sample: 'newcity', x: 32, z: 32, zoom: 1.2, W: 1440, H: 1000, mob: 0 },
];
const only = process.argv[2] ? process.argv[2].split(',') : null;
const WX = [
  { id: 'clear', q: 'season=1' },
  { id: 'rain', q: 'season=1&wx=1&wxTime=3.2' },
  { id: 'storm', q: 'season=1&wx=2&wxTime=3.2' },
  { id: 'storm-flash', q: 'season=1&wx=2&wxTime=3.2&flash=0.9' },
  { id: 'rainbow', q: 'season=0&rainbow=1' },
  { id: 'snow', q: 'season=3&wx=3&wxTime=3.2' },
  { id: 'blizzard', q: 'season=3&wx=4&wxTime=3.2' },
];
for (const v of VIEWS) {
  if (only && !only.includes(v.id)) continue;
  for (const w of WX) {
    const out = `${OUT}/${v.id}-${w.id}.png`;
    if (fs.existsSync(out) && !process.env.REDO) continue;
    const r = spawnSync('node', [SP + '/look-at-exp.mjs', v.sample, v.x, v.z, v.zoom, v.W, v.H, v.mob, out, '0', 'landArt=2&' + w.q], { env: { ...process.env, CHROME_PATH: '/opt/pw-browsers/chromium' }, encoding: 'utf8', timeout: 180000 });
    console.log(v.id, w.id, (r.stdout || '').trim().split('\n').pop(), r.status !== 0 ? 'ERR ' + (r.stderr || '').slice(-200) : '');
  }
}
console.log('done');
